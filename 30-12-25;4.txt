 #include<stdio.h>
 int main()
 {
      char ch;
      scanf("%ch",&ch);
      
      if (ch>='A'&&ch<='Z'){ 
          printf("it is upper case");
      }
        else if (ch>='a'&&ch<='z')
       {
          printf("it is lower case");
      }
      else {
          printf("no alphabets");
      }
      return 0;
 }
