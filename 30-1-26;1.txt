#include <stdio.h>

int main() {
    int a[10], n, i, count = 0;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter array elements:\n");
    for(i = 0; i < n; i++) {
        scanf("%d", &a[i]);
        if(a[i] < 0) {
            count++;
        }
    }

    printf("Negative numbers count = %d", count);

    return 0;
}
