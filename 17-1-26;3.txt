#include <stdio.h>

int main() {
    int n, a[50], count[50] = {0};
    int i, j;

    scanf("%d", &n);
    for(i = 0; i < n; i++)
        scanf("%d", &a[i]);

    for(i = 0; i < n; i++) {
        if(count[i] != -1) {
            int c = 1;
            for(j = i + 1; j < n; j++) {
                if(a[i] == a[j]) {
                    c++;
                    count[j] = -1;
                }
            }
            count[i] = c;
        }
    }

    for(i = 0; i < n; i++) {
        if(count[i] != -1)
            printf("%d->%d ", a[i], count[i]);
    }

    return 0;
}
