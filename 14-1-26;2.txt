 #include<stdio.h>
 int main()
 {
     int a,b,i,lcm;
     char result[45];
     scanf("%d%d",&a,&b);
     for(i=0;i<=a*b;i++){
         if(i%a==0&&i%b==0){
             lcm=i;
             break;
             
         }
     }
     sprintf(result,"%d",lcm);
     printf("%s",result);
     return 0;
 }
