#include<stdio.h>
int main()
{
    int num,digit,rev=0,temp;
    scanf("%d",&num);
    temp=num;
    while (num!=0){
        digit=num%10;
        rev=rev*10+digit;
        num=num/10;
    }
        if (temp==rev)
        printf("it  is palidrome");
        else
        printf("it is not palidrome");
        
    
    return 0;
    
}
