 #include<stdio.h>
 int main()
 {
     int num,digit,temp,sum=0;
     scanf("%d",&num);
     temp=num;
     
     while(num!=0)
     {
        
       digit=num%10;
       sum=sum+(digit*digit*digit);
       num=num/10;
       
     }
     
     if(sum==temp)
     printf(" aermstrong num");
     else 
     printf("not armstrong num");
     return 0;
 }
