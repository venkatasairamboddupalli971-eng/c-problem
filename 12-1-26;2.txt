#include <stdio.h>

int main() 
{
 char ch[78];
 int i,lower=0,upper=0;
 scanf("%s",ch);
 for (i=0;ch[i]!='\0';i++){
     if(ch[i]>='A' && ch[i]<='Z')
     upper++;
      
     else  if (ch[i]>='a' && ch[i]<='z')
     
     lower++;
 }
     printf("uppercase=%d\n",upper);
     printf("lowercase=%d",lower);
         
     
 
    return 0;
    
 }
