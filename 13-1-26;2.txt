#include <stdio.h>
#include<string.h>
 int main()
 {
char str[100];
int i,digit,sum=0;
int num=0;
 fgets(str,100,stdin);
 int len=strlen(str)-1;
 for(i=0;i<len;i++)
 {
     digit=str[i]-'0';
     sum+=digit*digit*digit;
     num=num*10+digit;
 }
 if(sum==num)
 printf("armstrong");
 else
 printf("not armstrong");
 return 0;
 }
